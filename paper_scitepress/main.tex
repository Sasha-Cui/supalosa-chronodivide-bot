% The official style asks for uppercase section headings. Preserve star and
% optional-title forms so the bibliography heading remains compatible.
\let\scitepresssection\section
\RenewDocumentCommand{\section}{s o m}{%
  \IfBooleanTF{#1}{%
    \IfNoValueTF{#2}{\scitepresssection*{\MakeUppercase{#3}}}{\scitepresssection*[{#2}]{\MakeUppercase{#3}}}%
  }{%
    \IfNoValueTF{#2}{\scitepresssection{\MakeUppercase{#3}}}{\scitepresssection[{#2}]{\MakeUppercase{#3}}}%
  }%
}

  \scitepressthebibliography{#1}%
  \setlength{\itemsep}{0pt}%
